# Öffentliche Regeln für werkzeuge/scrub/pruefen.py
# Eine Regex je Zeile, Groß-/Kleinschreibung egal; "CS:" = mit Groß-/Kleinschreibung;
# "HART:" = gilt auch in Zeilen mit scrub:erlaubt (echte Schlüssel lassen sich nicht wegmarkieren).
# Platzhalter wie C:\Users\<name>\ sind erlaubt (das "<" nach dem Pfadtrenner).

# Persönliche Pfade
\b[A-Za-z]:[\\/]{1,2}Users[\\/]{1,2}(?!<|\{|\$|Public\b)
(^|[\s"'(=])/c/Users/(?!<|\{|\$)
(^|[\s"'(=])/home/(?!<|\{|\$|runner\b|user\b|app\b|node\b|pwuser\b)[a-z][a-z0-9_-]+/
(^|[\s"'(=])/Users/(?!<|\{|\$|Shared\b)[A-Za-z][A-Za-z0-9_-]+/
\b[D-Z]:[\\/]{1,2}(?!<|\{)[A-Za-z]

# Private Netze und Geräte
\b10\.\d{1,3}\.\d{1,3}\.\d{1,3}\b
\b192\.168\.\d{1,3}\.\d{1,3}\b
\b172\.(1[6-9]|2\d|3[01])\.\d{1,3}\.\d{1,3}\b
\b100\.(6[4-9]|[7-9]\d|1[01]\d|12[0-7])\.\d{1,3}\.\d{1,3}\b
\b([0-9A-F]{2}:){5}[0-9A-F]{2}\b
\bf[cd][0-9a-f]{2}:[0-9a-f]{1,4}:[0-9a-f:]+
\b[a-z0-9-]+\.[a-z0-9-]+\.ts\.net\b

# E-Mail-Adressen außer Beispiel-, noreply- und Projektadressen (dazu GitHubs eigene Absender: Merge- und
# Dependabot-Commits tragen noreply@github.com und support@github.com)
\b[A-Za-z0-9._%+-]+@(?!example\.(org|com|net)\b|beispiel\.test\b|(users\.)?noreply\.github\.com\b|github\.com\b|anthropic\.com\b|localhost\b)[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b
